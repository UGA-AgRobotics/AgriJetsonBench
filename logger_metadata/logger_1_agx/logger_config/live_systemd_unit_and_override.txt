# /etc/systemd/system/pc_external_logger@.service
[Unit]
Description=PC external sensor logger + live dashboard (Pc_package)
After=network.target

[Service]
Type=simple
User=%i
WorkingDirectory=/home/%i/Pc_package
Environment=PC_ENV_MODE=venv
Environment=PYTHON_VENV_PATH=/home/%i/envs/pc_logger
Environment=PC_SENSOR_PORT=/dev/ttyACM0
Environment=PC_CURRENT_SIGN=1
ExecStart=/home/%i/Pc_package/scripts/run_pc_logger.sh
Restart=on-failure
RestartSec=3

[Install]
WantedBy=multi-user.target
